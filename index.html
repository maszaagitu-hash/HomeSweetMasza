function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetPribadi = ss.getSheetByName("Pribadi"); var sheetLove = ss.getSheetByName("Love"); var txPribadi = []; if (sheetPribadi) { var dataPribadi = sheetPribadi.getDataRange().getValues(); for (var i = 1; i < dataPribadi.length; i++) { if (dataPribadi[i][0]) { txPribadi.push({ date: formatDate(dataPribadi[i][0]), type: dataPribadi[i][1], account: dataPribadi[i][2], category: dataPribadi[i][3], desc: dataPribadi[i][4], amount: Number(dataPribadi[i][5]) || 0 }); } } } var txLove = []; if (sheetLove) { var dataLove = sheetLove.getDataRange().getValues(); for (var j = 1; j < dataLove.length; j++) { if (dataLove[j][0]) { txLove.push({ date: formatDate(dataLove[j][0]), penyetor: dataLove[j][1], account: dataLove[j][2], desc: dataLove[j][3], amount: Number(dataLove[j][4]) || 0 }); } } } return ContentService.createTextOutput(JSON.stringify({ txPribadi: txPribadi, txLove: txLove })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var params = JSON.parse(e.postData.contents); if (params.action === "addPribadi") { var sheet = ss.getSheetByName("Pribadi"); if (!sheet) sheet = ss.insertSheet("Pribadi"); sheet.appendRow([params.date, params.type, params.account, params.category, params.desc, params.amount]); } else if (params.action === "addLove") { var sheet = ss.getSheetByName("Love"); if (!sheet) sheet = ss.insertSheet("Love"); sheet.appendRow([params.date, params.penyetor, params.account, params.desc, params.amount]); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function formatDate(d) { if (d instanceof Date) { return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd"); } return String(d); }
